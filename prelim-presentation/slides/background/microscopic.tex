\begin{frame}
  \frametitle{The microscopic approach irradiates a sample computationally}
  \textbf{The delayed neutron count rate is calculated from data and concentrations.}

  \begin{equation}
    \dot{n}_{d}(t) = \sum_{i}^{I} P_{n, i} \lambda_i N_i(t),
    \label{eq:dnpsingleeq}
  \end{equation}
  \begin{align}
    \intertext{where}
    I &= \text{ the number of \glspl{DNP} [$\#$]} \nonumber \\
    \dot{n}_{d} &= \mbox{ the delayed neutron count rate $\left[\# \cdot s^{-1}\right]$} \nonumber \\
    P_{n, i} &= \mbox{ the average number of delayed neutrons emitted from} \nonumber\\
    & \;\;\;\;\;\; \mbox{decay of nuclide $i$, or the emission probability [$\#$]} \nonumber \\
    \lambda_i &= \mbox{ the decay constant of \gls{DNP} $i$ $[s^{-1}]$} \nonumber \\
    N_i(t) &= \mbox{ the number of atoms of \gls{DNP} $i$ $\left[\#\right]$.} \nonumber
  \end{align}
\end{frame}
