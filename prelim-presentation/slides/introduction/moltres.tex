\begin{frame}
  \frametitle{Moltres provides multiphysics \gls{MSR} simulation capability}
        Moltres is a multiphysics reactor simulation tool built on the MOOSE framework \cite{lindsay_introduction_2018}.

        \begin{block}{Moltres enables:}
            \begin{itemize}
                \item Multigroup neutron diffusion
                \item \Gls{DNP} recirculation
                \item Heat transfer and fluid flow
            \end{itemize}
        \end{block}

        \textbf{Moltres will be used for the proposed safety, security, and safeguards transient simulations.}
\end{frame}
