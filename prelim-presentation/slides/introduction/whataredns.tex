  \frametitle{What are Delayed Neutrons?}
        \Glspl{DNP} are nuclides that undergo $\beta^-$ decay, producing a daughter, called an emitter, which emits a delayed neutron \cite{bank2000jef}.

        \begin{block}{Relevant nuclear data for \glspl{DNP} includes:}
            \[
            \begin{aligned}
                P_{n, i} &= \text{the emission probability [-]} \\
                \tau_i &= \text{the half-life of \gls{DNP} $i$ [$s$]} \\
                FY_i &= \text{the fission yield of $i$ [-]}.
            \end{aligned}
            \]
        \end{block}

        \textbf{Accurately modeling the \glspl{DNP} is important for transient simulations.}
