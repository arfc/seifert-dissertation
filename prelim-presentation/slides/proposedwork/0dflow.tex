\begin{frame}
  \frametitle{The 0D flow model will introduce new features and analyses}
    \begin{block}{New features in the 0D flow model:}
        \begin{itemize}
            \item Full depletion chain modeling in OpenMC
            \begin{itemize}
                \item Improving chemical removal modeling
                \item Enabling residence time analysis
                \item Enabling non-equilibrium \gls{DNP} concentrations
                \item Improving \gls{DNP} concentration accuracy
            \end{itemize}
	\end{itemize}
	\end{block}
	\begin{block}{New analyses using the 0D flow model:}
	\begin{itemize}
            \item Novel analysis for residual minimization using pulse and saturation irradiation data \cite{brady1988evaluation, keepin1957delayed}
            \item Comparison of various nuclear datasets \cite{huynh2014calculation, bank2000jef, mathieu2012new}
            \item Calculation of the improved \gls{DNP} group spectra ($\chi_{d,k}$)
        \end{itemize}
    \end{block}
\end{frame}
