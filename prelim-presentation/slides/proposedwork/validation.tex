  \frametitle{The MSRE will be used for validation}
    \begin{block}{Necessary components will be computed/generated:}
        \begin{itemize}
            \item Improved \gls{DNP} group parameters for the \gls{MSRE}
            \item Multi-group cross sections using OpenMC for the \gls{MSRE}
            \item Upper and lower plena added to an existing \gls{MSRE} Moltres model
        \end{itemize}
    \end{block}
    \begin{block}{Quantities of interest will be computed and compared with experimental data for validation:}
      \begin{itemize}
          \item Calculation of $\beta_{eff}$ using the prompt method for stationary and flowing fuel salt using improved and conventional \gls{DNP} group parameters \cite{haubenreich1970experience}
          \item Simulate pump start-up and coast-down transients to compare reactivity responses to experimental data \cite{prince1968zero}
      \end{itemize}
    \end{block}
