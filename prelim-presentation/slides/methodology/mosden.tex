\begin{frame}
  \frametitle{MoSDeN processes data and generates improved groups}
	\Gls{MoSDeN} is an open-source Python package in development that will enable users to generate improved \gls{DNP} groups.
	\begin{block}{MoSDeN currently enables:}
		\begin{itemize}
			\item Use of various data inputs for half-lives ($\tau$), emission probabilities ($P_n$), and cumulative fission yields (CFY)
			\begin{itemize}
				\item ENDF-6 formatted data
				\item OpenMC chain data
				\item Custom data
			\end{itemize}
			\item Calculation of concentrations and delayed neutron count rates of an irradiated sample
			\item Calculation of the \gls{DNP} group parameters that fit the calculated count rate
		\end{itemize}
	\end{block}

	\textbf{The MoSDeN package is in development and offers improved \gls{DNP} group generation for \glspl{MSR} on a per-problem basis.}
\end{frame}
