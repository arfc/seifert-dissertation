  \frametitle{The saturation irradiation is a non-linear least squares problem}
  \textbf{The group parameters are calculated by equating the delayed neutron count rates.}

  \begin{equation}
    \textcolor{orange}{\dot{F}_s} \sum_{k=1}^{K}  \textcolor{olive}{\bar{\nu}_{d, k}} \textcolor{orange}{e^{-\lambda_k t}} \approx \sum_{i}^{I} \textcolor{blue}{P_{n, i} \lambda_i N_i(t)}
  \end{equation}

    \begin{equation}
    \textcolor{orange}{
    \dot{F_s}
    \begin{bmatrix}
    e^{-\lambda_1 t_0} & \hdots &  e^{-\lambda_K t_0}\\
    \vdots                                     & \ddots &  \vdots                      \\
    e^{-\lambda_1 t_m} &  \hdots  & e^{-\lambda_K t_m}\\
    \end{bmatrix}
    }
    \textcolor{olive}{
    \begin{bmatrix}
    \bar{\nu}_{d, 1}\\
    \vdots\\
    \bar{\nu}_{d, K}\\

    \dot{n}_d(t_0)  \\
    \vdots\\

    \end{bmatrix}
    }
    \label{eq:lambda}
    \end{equation}
